From idea to trajectory

A few lines from equations to a plot

Define the initial conditions, pick a metric and let EinsteinPy integrate the geodesic.

  geodesic.py
import numpy as np
from einsteinpy.geodesic import Timelike
from einsteinpy.plotting.geodesic import StaticGeodesicPlotter

geod = Timelike(
    metric="Schwarzschild",
    metric_params=(0.,),
    position=[40., np.pi / 2, 0.],
    momentum=[0., 0., 3.83405],
    steps=5500,
    delta=1,
    return_cartesian=True,
)

plotter = StaticGeodesicPlotter()
plotter.plot(geod)
plotter.show()
