Symbolic relativity

Curvature, derived exactly

Build a metric, then get Christoffel symbols, Riemann, Ricci, Einstein and Weyl tensors as SymPy expressions you can simplify, differentiate and lambdify.

  symbolic.py
from sympy import simplify
from einsteinpy.symbolic import (
    Schwarzschild, ChristoffelSymbols,
    RicciTensor, RicciScalar, EinsteinTensor,
)

m = Schwarzschild()            # predefined metric

ch = ChristoffelSymbols.from_metric(m)
simplify(ch[0, 0, 1])   # r_s / (2*r*(r - r_s))
simplify(ch[1, 2, 2])   # -r + r_s
simplify(ch[2, 3, 3])   # -sin(2*theta)/2

ricci = RicciTensor.from_metric(m)       # every component -> 0
R     = RicciScalar.from_metric(m)       # -> 0
G     = EinsteinTensor.from_metric(m)    # -> 0 (vacuum)
